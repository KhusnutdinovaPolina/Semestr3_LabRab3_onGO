import (
        "encoding/gob"
        "encoding/json"
        "fmt"
        "os"
)

type Array struct {
        Data []string `json:"data"`
}

// Конструктор
func NewArray() *Array {
        return &Array{Data: []string{}}
}

// Добавить в конец
func (a *Array) PushBack(val string) {
        a.Data = append(a.Data, val)
}

// Вставить по индексу (аналог insertAt)
func (a *Array) InsertAt(index int, val string) {
        if index < 0 || index > len(a.Data) {
                return
        }
        // Вставляем элемент в середину через срезы
        a.Data = append(a.Data[:index], append([]string{val}, a.Data[index:]...)...)
}

// Получить по индексу
func (a *Array) Get(index int) string {
        if index < 0 || index >= len(a.Data) {
                return ""
        }
        return a.Data[index]
}

// Заменить по индексу
func (a *Array) Set(index int, val string) {
        if index >= 0 && index < len(a.Data) {
                a.Data[index] = val
        }
}

// Удалить по индексу
func (a *Array) RemoveAt(index int) {
        if index >= 0 && index < len(a.Data) {
                a.Data = append(a.Data[:index], a.Data[index+1:]...)
        }
}

// Длина
func (a *Array) Length() int {
        return len(a.Data)
}

// Очистить
func (a *Array) Clear() {
        a.Data = []string{}
}

// Печать
func (a *Array) Print() {
        fmt.Println(a.Data)
}

// Сохранение (JSON)
func (a *Array) SaveToFile(filename string) error {
        data, _ := json.MarshalIndent(a, "", "    ")
        return os.WriteFile(filename, data, 0644)
}

// Загрузка (JSON)
func (a *Array) LoadFromFile(filename string) error {
        data, err := os.ReadFile(filename)
        if err != nil {
                return err
        }
        return json.Unmarshal(data, a)
}

// Бинарное сохранение (Gob)
func (a *Array) SaveToBinary(filename string) error {
        file, _ := os.Create(filename)
        defer file.Close()
        encoder := gob.NewEncoder(file)
        return encoder.Encode(a)
}

// Бинарная загрузка (Gob)
func (a *Array) LoadFromBinary(filename string) error {
        file, err := os.Open(filename)
        if err != nil {
                return err
        }
        defer file.Close()
        decoder := gob.NewDecoder(file)
        return decoder.Decode(a)
}

import (

type TNode struct {
        Data  string `json:"data"`
        Left  *TNode `json:"left"`
        Right *TNode `json:"right"`
}

type CBT struct {
        Root  *TNode `json:"root"`
        Count int    `json:"count"`
}

// Конструктор
func NewCBT() *CBT {
        return &CBT{}
}

// Вспомогательный метод: поиск узла по индексу
func (t *CBT) getNodeByIndex(node *TNode, currentIndex, targetIndex int) *TNode {
        if node == nil {
                return nil
        }
        if currentIndex == targetIndex {
                return node
        }
        // Ищем в левом поддереве
        leftRes := t.getNodeByIndex(node.Left, 2*currentIndex+1, targetIndex)
        if leftRes != nil {
                return leftRes
        }
        // Если слева нет, ищем в правом
        return t.getNodeByIndex(node.Right, 2*currentIndex+2, targetIndex)
}

// Вставка нового элемента
func (t *CBT) Insert(val string) {
        newNode := &TNode{Data: val}
        if t.Count == 0 {
                t.Root = newNode
                t.Count = 1
                return
        }
        newIndex := t.Count
        parentIndex := (newIndex - 1) / 2
        isLeft := (newIndex%2 == 1)
        parent := t.getNodeByIndex(t.Root, 0, parentIndex)

        if isLeft {
                parent.Left = newNode
        } else {
                parent.Right = newNode
        }
        t.Count++
}

// Проверка на Complete
func (t *CBT) IsComplete() bool {
        if t.Root == nil {
                return true
        }

        // Используем массив как очередь
        queue := make([]*TNode, 0)
        queue = append(queue, t.Root)
        seenNull := false

        for len(queue) > 0 {
                curr := queue[0]
                queue = queue[1:] // Dequeue

                if curr != nil {
                        if seenNull {
                                return false // Нашли узел после пустоты
                        }
                        queue = append(queue, curr.Left)
                        queue = append(queue, curr.Right)
                } else {
                        seenNull = true
                }
        }
        return true
}

// Размер дерева
func (t *CBT) Size() int {
        return t.Count
}

// Вспомогательный метод для очистки
func (t *CBT) clearRecursive(node *TNode) {
        if node == nil {
                return
        }
        t.clearRecursive(node.Left)
        t.clearRecursive(node.Right)
        node = nil
}

// Очистка
func (t *CBT) Clear() {
        t.clearRecursive(t.Root)
        t.Root = nil
        t.Count = 0
}

// Вспомогательный метод для печати
func (t *CBT) printRecursive(node *TNode, prefix string, isLast bool) {
        if node == nil {
                return
        }
        fmt.Printf("%s", prefix)
        if isLast {
                fmt.Print("+-- ")
        } else {
                fmt.Print("|-- ")
        }
        fmt.Println(node.Data)

        newPrefix := prefix
        if isLast {
                newPrefix += "    "
        } else {
                newPrefix += "|   "
        }

        hasL := node.Left != nil
        hasR := node.Right != nil
        if hasL {
                t.printRecursive(node.Left, newPrefix, !hasR)
        }
        if hasR {
                t.printRecursive(node.Right, newPrefix, true)
        }
}

// Печать дерева
func (t *CBT) Print() {
        if t.Root == nil {
                fmt.Println("(дерево пусто)")
                return
        }
        t.printRecursive(t.Root, "", true)
}

// Сохранение (JSON)
func (t *CBT) SaveToFile(filename string) error {
        data, _ := json.MarshalIndent(t, "", "    ")
        return os.WriteFile(filename, data, 0644)
}

// Загрузка (JSON)
func (t *CBT) LoadFromFile(filename string) error {
        data, err := os.ReadFile(filename)
        if err != nil {
                return err
        }
        return json.Unmarshal(data, t)
}
func (t *CBT) SaveToBinary(filename string) error {
        file, _ := os.Create(filename)
        defer file.Close()
        encoder := gob.NewEncoder(file)
        return encoder.Encode(t)
}

func (t *CBT) LoadFromBinary(filename string) error {
        file, err := os.Open(filename)
        if err != nil {
                return err
        }
        defer file.Close()
        decoder := gob.NewDecoder(file)
        return decoder.Decode(t)
}

const SIZE = 10
const MAX_KICKS = 20

type CuckooHashTable struct {
        Keys1 [SIZE]string `json:"keys1"`
        Vals1 [SIZE]string `json:"vals1"`
        Keys2 [SIZE]string `json:"keys2"`
        Vals2 [SIZE]string `json:"vals2"`
}

// Конструктор
func NewCuckooHashTable() *CuckooHashTable {
        return &CuckooHashTable{}
}

// Первая хеш-функция
func (c *CuckooHashTable) hash1(key string) int {
        sum := 0
        for _, ch := range key {
                sum += int(ch)
        }
        return sum % SIZE
}

// Вторая хеш-функция
func (c *CuckooHashTable) hash2(key string) int {
        sum := 0
        for _, ch := range key {
                sum += int(ch) * 7
        }
        return sum % SIZE
}

// Вставка (с выбиванием)
func (c *CuckooHashTable) Insert(key, value string) bool {
        // Если ключ уже есть в первой таблице — обновляем
        if c.Keys1[c.hash1(key)] == key {
                c.Vals1[c.hash1(key)] = value
                return true
        }
        // Если во второй — обновляем
        if c.Keys2[c.hash2(key)] == key {
                c.Vals2[c.hash2(key)] = value
                return true
        }

        curKey, curVal := key, value
        for i := 0; i < MAX_KICKS; i++ {
                // Пробуем вставить в таблицу 1
                pos1 := c.hash1(curKey)
                if c.Keys1[pos1] == "" {
                        c.Keys1[pos1], c.Vals1[pos1] = curKey, curVal
                        return true
                }
                // Место занято — выбиваем
                curKey, c.Keys1[pos1] = c.Keys1[pos1], curKey
                curVal, c.Vals1[pos1] = c.Vals1[pos1], curVal

                // Пробуем вставить в таблицу 2
                pos2 := c.hash2(curKey)
                if c.Keys2[pos2] == "" {
                        c.Keys2[pos2], c.Vals2[pos2] = curKey, curVal
                        return true
                }
                // Место занято — выбиваем
                curKey, c.Keys2[pos2] = c.Keys2[pos2], curKey
                curVal, c.Vals2[pos2] = c.Vals2[pos2], curVal
        }
        return false // Переполнение
}

// Поиск по ключу
func (c *CuckooHashTable) Get(key string) string {
        pos1 := c.hash1(key)
        if c.Keys1[pos1] == key {
                return c.Vals1[pos1]
        }
        pos2 := c.hash2(key)
        if c.Keys2[pos2] == key {
                return c.Vals2[pos2]
        }
        return "" // Не найдено
}

// Печать обеих таблиц
func (c *CuckooHashTable) Print() {
        fmt.Println("--- Cuckoo (Table 1) ---")
        for i := 0; i < SIZE; i++ {
                if c.Keys1[i] != "" {
                        fmt.Printf("[%d]: %s=%s\n", i, c.Keys1[i], c.Vals1[i])
                }
        }
        fmt.Println("--- Cuckoo (Table 2) ---")
        for i := 0; i < SIZE; i++ {
                if c.Keys2[i] != "" {
                        fmt.Printf("[%d]: %s=%s\n", i, c.Keys2[i], c.Vals2[i])
                }
        }
}

// Сохранение (JSON)
func (c *CuckooHashTable) SaveToFile(filename string) error {
        data, _ := json.MarshalIndent(c, "", "    ")
        return os.WriteFile(filename, data, 0644)
}

import (

type DNode struct {
        Data string `json:"data"`
        Prev *DNode `json:"prev"`
        Next *DNode `json:"next"`
}

type DList struct {
        Head *DNode `json:"head"`
        Tail *DNode `json:"tail"`
        Size int    `json:"size"`
}

func NewDList() *DList { return &DList{} }

func (l *DList) PushHead(val string) {
        newNode := &DNode{Data: val, Next: l.Head}
        if l.Head != nil {
                l.Head.Prev = newNode
        }
        l.Head = newNode
        if l.Tail == nil {
                l.Tail = newNode
        }
        l.Size++
}

func (l *DList) PushTail(val string) {
        newNode := &DNode{Data: val, Prev: l.Tail}
        if l.Tail != nil {
                l.Tail.Next = newNode
        }
        l.Tail = newNode
        if l.Head == nil {
                l.Head = newNode
        }
        l.Size++
}

// Вставить после
func (l *DList) InsertAfter(target, val string) {
        curr := l.Head
        for curr != nil && curr.Data != target {
                curr = curr.Next
        }
        if curr == nil {
                return
        }
        newNode := &DNode{Data: val, Prev: curr, Next: curr.Next}
        if curr.Next != nil {
                curr.Next.Prev = newNode
        } else {
                l.Tail = newNode
        }
        curr.Next = newNode
        l.Size++
}

// Вставить перед
func (l *DList) InsertBefore(target, val string) {
        curr := l.Head
        for curr != nil && curr.Data != target {
                curr = curr.Next
        }
        if curr == nil {
                return
        }
        newNode := &DNode{Data: val, Prev: curr.Prev, Next: curr}
        if curr.Prev != nil {
                curr.Prev.Next = newNode
        } else {
                l.Head = newNode
        }
        curr.Prev = newNode
        l.Size++
}

// Удалить после
func (l *DList) PopAfter(target string) {
        curr := l.Head
        for curr != nil && curr.Data != target {
                curr = curr.Next
        }
        if curr == nil || curr.Next == nil {
                return
        }
        toDelete := curr.Next
        curr.Next = toDelete.Next
        if toDelete.Next != nil {
                toDelete.Next.Prev = curr
        } else {
                l.Tail = curr
        }
        l.Size--
}

// Удалить перед
func (l *DList) PopBefore(target string) {
        curr := l.Head
        for curr != nil && curr.Data != target {
                curr = curr.Next
        }
        if curr == nil || curr.Prev == nil {
                return
        }
        toDelete := curr.Prev
        curr.Prev = toDelete.Prev
        if toDelete.Prev != nil {
                toDelete.Prev.Next = curr
        } else {
                l.Head = curr
        }
        l.Size--
}

func (l *DList) PopHead() {
        if l.Head == nil {
                return
        }
        l.Head = l.Head.Next
        if l.Head != nil {
                l.Head.Prev = nil
        } else {
                l.Tail = nil
        }
        l.Size--
}

func (l *DList) PopTail() {
        if l.Tail == nil {
                return
        }
        l.Tail = l.Tail.Prev
        if l.Tail != nil {
                l.Tail.Next = nil
        } else {
                l.Head = nil
        }
        l.Size--
}

func (l *DList) Find(val string) bool {
        curr := l.Head
        for curr != nil {
                if curr.Data == val {
                        return true
                }
                curr = curr.Next
        }
        return false
}

func (l *DList) Get(index int) string {
        if index < 0 || index >= l.Size {
                return ""
        }
        curr := l.Head
        for i := 0; i < index; i++ {
                curr = curr.Next
        }
        return curr.Data
}

func (l *DList) Length() int { return l.Size }
func (l *DList) Clear()      { l.Head = nil; l.Tail = nil; l.Size = 0 }

func (l *DList) PrintForward() {
        curr := l.Head
        fmt.Print("[")
        for curr != nil {
                fmt.Print(curr.Data)
                if curr.Next != nil {
                        fmt.Print(" <-> ")
                }
                curr = curr.Next
        }
        fmt.Println("]")
}

func (l *DList) PrintBackward() {
        curr := l.Tail
        fmt.Print("[")
        for curr != nil {
                fmt.Print(curr.Data)
                if curr.Prev != nil {
                        fmt.Print(" <-> ")
                }
                curr = curr.Prev
        }
        fmt.Println("]")
}

func (l *DList) SaveToFile(filename string) error {
        data, _ := json.MarshalIndent(l, "", "    ")
        return os.WriteFile(filename, data, 0644)
}

func (l *DList) LoadFromFile(filename string) error {
        data, err := os.ReadFile(filename)
        if err != nil {
                return err
        }
        return json.Unmarshal(data, l)
}

// Бинарное сохранение (Gob)
func (l *DList) SaveToBinary(filename string) error {
        file, _ := os.Create(filename)
        defer file.Close()
        encoder := gob.NewEncoder(file)
        return encoder.Encode(l)
}

type HashTable struct {
        Table [10][]string `json:"table"`
}

// Конструктор
func NewHashTable() *HashTable {
        ht := &HashTable{}
        for i := range ht.Table {
                ht.Table[i] = []string{}
        }
        return ht
}

// Хеш-функция
func (h *HashTable) hash(key string) int {
        sum := 0
        for _, c := range key {
                sum += int(c)
        }
        return sum % 10
}

// Вставка / Обновление
func (h *HashTable) Insert(key, value string) {
        idx := h.hash(key)
        entry := key + ":" + value

        // Проверяем, есть ли уже такой ключ (обновление)
        for i, item := range h.Table[idx] {
                if strings.HasPrefix(item, key+":") {
                        h.Table[idx][i] = entry
                        return
                }
        }
        // Если нет — добавляем
        h.Table[idx] = append(h.Table[idx], entry)
}

// Получить значение по ключу
func (h *HashTable) Get(key string) string {
        idx := h.hash(key)
        for _, item := range h.Table[idx] {
                if strings.HasPrefix(item, key+":") {
                        return strings.TrimPrefix(item, key+":")
                }
        }
        return ""
}

// Удалить по ключу
func (h *HashTable) Remove(key string) {
        idx := h.hash(key)
        for i, item := range h.Table[idx] {
                if strings.HasPrefix(item, key+":") {
                        // Удаляем элемент из среза
                        h.Table[idx] = append(h.Table[idx][:i], h.Table[idx][i+1:]...)
                        return
                }
        }
}

// Печать таблицы
func (h *HashTable) Print() {
        fmt.Println("--- HashTable ---")
        for i, bucket := range h.Table {
                fmt.Printf("[%d]: %v\n", i, bucket)
        }
}

// Сохранение (JSON)
func (h *HashTable) SaveToFile(filename string) error {
        data, _ := json.MarshalIndent(h, "", "    ")
        return os.WriteFile(filename, data, 0644)
}

// Загрузка (JSON)
func (h *HashTable) LoadFromFile(filename string) error {
        data, err := os.ReadFile(filename)
        if err != nil {
                return err
        }
        return json.Unmarshal(data, h)
}

func (h *HashTable) SaveToBinary(filename string) error {
        file, _ := os.Create(filename)
        defer file.Close()
        encoder := gob.NewEncoder(file)
        return encoder.Encode(h)
}

import (

type SNode struct {
        Data string `json:"data"`
        Next *SNode `json:"next"`
}

type SList struct {
        Head *SNode `json:"head"`
        Size int    `json:"size"`
}

func NewSList() *SList { return &SList{} }

func (l *SList) PushHead(val string) {
        l.Head = &SNode{Data: val, Next: l.Head}
        l.Size++
}

func (l *SList) PushTail(val string) {
        newNode := &SNode{Data: val}
        if l.Head == nil {
                l.Head = newNode
        } else {
                curr := l.Head
                for curr.Next != nil {
                        curr = curr.Next
                }
                curr.Next = newNode
        }
        l.Size++
}

func (l *SList) PopHead() {
        if l.Head == nil {
                return
        }
        l.Head = l.Head.Next
        l.Size--
}

func (l *SList) PopTail() {
        if l.Head == nil {
                return
        }
        if l.Head.Next == nil {
                l.Head = nil
                l.Size--
                return
        }
        curr := l.Head
        for curr.Next.Next != nil {
                curr = curr.Next
        }
        curr.Next = nil
        l.Size--
}

// Вставить после узла target
func (l *SList) InsertAfter(target, val string) {
        curr := l.Head
        for curr != nil && curr.Data != target {
                curr = curr.Next
        }
        if curr == nil {
                return
        }
        newNode := &SNode{Data: val, Next: curr.Next}
        curr.Next = newNode
        l.Size++
}

// Вставить перед узлом target
func (l *SList) InsertBefore(target, val string) {
        if l.Head == nil {
                return
        }
        if l.Head.Data == target {
                l.PushHead(val)
                return
        }
        curr := l.Head
        for curr.Next != nil && curr.Next.Data != target {
                curr = curr.Next
        }
        if curr.Next == nil {
                return
        }
        newNode := &SNode{Data: val, Next: curr.Next}
        curr.Next = newNode
        l.Size++
}

// Удалить ПОСЛЕ узла target
func (l *SList) PopAfter(target string) {
        curr := l.Head
        for curr != nil && curr.Data != target {
                curr = curr.Next
        }
        if curr == nil || curr.Next == nil {
                return
        }
        curr.Next = curr.Next.Next
        l.Size--
}

// Удалить ПЕРЕД узлом target
func (l *SList) PopBefore(target string) {
        if l.Head == nil || l.Head.Data == target {
                return
        }
        if l.Head.Next != nil && l.Head.Next.Data == target {
                l.PopHead()
                return
        }
        curr := l.Head
        for curr.Next != nil && curr.Next.Next != nil && curr.Next.Next.Data != target {
                curr = curr.Next
        }
        if curr.Next == nil || curr.Next.Next == nil {
                return
        }
        curr.Next = curr.Next.Next
        l.Size--
}

func (l *SList) Find(val string) bool {
        curr := l.Head
        for curr != nil {
                if curr.Data == val {
                        return true
                }
                curr = curr.Next
        }
        return false
}

func (l *SList) Get(index int) string {
        if index < 0 || index >= l.Size {
                return ""
        }
        curr := l.Head
        for i := 0; i < index; i++ {
                curr = curr.Next
        }
        return curr.Data
}

func (l *SList) Length() int { return l.Size }
func (l *SList) Clear()      { l.Head = nil; l.Size = 0 }

func (l *SList) Print() {
        curr := l.Head
        fmt.Print("[")
        for curr != nil {
                fmt.Print(curr.Data)
                if curr.Next != nil {
                        fmt.Print(" -> ")
                }
                curr = curr.Next
        }
        fmt.Println("]")
}

func (l *SList) SaveToFile(filename string) error {
        data, _ := json.MarshalIndent(l, "", "    ")
        return os.WriteFile(filename, data, 0644)
}

func (l *SList) LoadFromFile(filename string) error {
        data, err := os.ReadFile(filename)
        if err != nil {
                return err
        }
        return json.Unmarshal(data, l)

}

        file, _ := os.Create(filename)

        encoder := gob.NewEncoder(file)
        return encoder.Encode(l)
}

        file, err := os.Open(filename)
        if err != nil {
                return err
        }

        Items []string `json:"items"`
}

        return &Stack{Items: []string{}}
}

// Добавить на вершину
func (s *Stack) Push(val string) {
        s.Items = append(s.Items, val)
}

// Снять с вершины (и вернуть)
func (s *Stack) Pop() string {
        if len(s.Items) == 0 {
                return ""
        }
        val := s.Items[len(s.Items)-1]
        s.Items = s.Items[:len(s.Items)-1]
        return val
}

// Посмотреть вершину
func (s *Stack) Top() string {
        if len(s.Items) == 0 {
                return ""
        }
        return s.Items[len(s.Items)-1]
}

// Проверить на пустоту
func (s *Stack) IsEmpty() bool {
        return len(s.Items) == 0
}

// Получить размер
func (s *Stack) GetSize() int {
        return len(s.Items)
}

// Очистить
func (s *Stack) Clear() {
        s.Items = []string{}
}

func (s *Stack) Print() {
        // Печатаем в обратном порядке, так как вершина в конце среза
        fmt.Print("[")

                if i > 0 {

}
